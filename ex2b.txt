# ================= HELPER FUNCTIONS =================

def xtime(a):
    return ((a << 1) ^ 0x1B) & 0xFF if a & 0x80 else (a << 1)

def mul(a, b):
    r = 0
    for _ in range(8):
        if b & 1:
            r ^= a
        a = xtime(a)
        b >>= 1
    return r & 0xFF

def print_state(state, title):
    print(title)
    for r in range(4):
        for c in range(4):
            print(f"{state[r][c]:3}", end=" ")
        print()
    print()

# ================= AES TRANSFORMS =================

def sub_bytes(s): return [[SBOX[b] for b in row] for row in s]
def inv_sub_bytes(s): return [[INV_SBOX[b] for b in row] for row in s]

def shift_rows(s):
    return [s[0], s[1][1:]+s[1][:1], s[2][2:]+s[2][:2], s[3][3:]+s[3][:3]]

def inv_shift_rows(s):
    return [s[0], s[1][-1:]+s[1][:-1], s[2][-2:]+s[2][:-2], s[3][-3:]+s[3][:-3]]

def mix_columns(s):
    for i in range(4):
        c = [s[r][i] for r in range(4)]
        t = c[0]^c[1]^c[2]^c[3]
        s[0][i] ^= t ^ xtime(c[0]^c[1])
        s[1][i] ^= t ^ xtime(c[1]^c[2])
        s[2][i] ^= t ^ xtime(c[2]^c[3])
        s[3][i] ^= t ^ xtime(c[3]^c[0])
    return s

def inv_mix_columns(s):
    for i in range(4):
        c = [s[r][i] for r in range(4)]
        s[0][i] = mul(c[0],14)^mul(c[1],11)^mul(c[2],13)^mul(c[3],9)
        s[1][i] = mul(c[0],9)^mul(c[1],14)^mul(c[2],11)^mul(c[3],13)
        s[2][i] = mul(c[0],13)^mul(c[1],9)^mul(c[2],14)^mul(c[3],11)
        s[3][i] = mul(c[0],11)^mul(c[1],13)^mul(c[2],9)^mul(c[3],14)
    return s

def add_round_key(s, k):
    return [[s[r][c]^k[r][c] for c in range(4)] for r in range(4)]

# ================= KEY EXPANSION =================

def key_expansion(key_bytes):
    if len(key_bytes) != 16:
        raise ValueError("Key must be exactly 16 bytes")

    kb = key_bytes[:]
    w = [kb[i:i+4] for i in range(0,16,4)]

    for i in range(4,44):
        t = w[i-1][:]
        if i % 4 == 0:
            t = t[1:]+t[:1]
            t = [SBOX[b] for b in t]
            t[0] ^= RCON[i//4-1]
        w.append([w[i-4][j]^t[j] for j in range(4)])

    return [[w[r+i] for i in range(4)] for r in range(0,44,4)]

# ================= AES ENCRYPT =================

def aes_encrypt(pt, key_bytes, verbose=True):
    if len(pt) != 16:
        raise ValueError("Plaintext must be exactly 16 bytes")

    s = [[ord(pt[r+4*c]) for c in range(4)] for r in range(4)]
    rk = key_expansion(key_bytes)

    if verbose: print_state(s, "Initial State")

    s = add_round_key(s, rk[0])
    if verbose: print_state(s, "After AddRoundKey (0)")

    for r in range(1,10):
        s = sub_bytes(s)
        s = shift_rows(s)
        s = mix_columns(s)
        s = add_round_key(s, rk[r])
        if verbose: print_state(s, f"After Round {r}")

    s = sub_bytes(s)
    s = shift_rows(s)
    s = add_round_key(s, rk[10])
    if verbose: print_state(s, "Final Ciphertext State")

    return s

# ================= AES DECRYPT =================

def aes_decrypt(c, key_bytes, verbose=True):
    rk = key_expansion(key_bytes)
    s = add_round_key(c, rk[10])

    for r in range(9,0,-1):
        s = inv_shift_rows(s)
        s = inv_sub_bytes(s)
        s = add_round_key(s, rk[r])
        s = inv_mix_columns(s)
        if verbose: print_state(s, f"After InvRound {r}")

    s = inv_shift_rows(s)
    s = inv_sub_bytes(s)
    s = add_round_key(s, rk[0])
    if verbose: print_state(s, "Recovered Plaintext State")

    return s
